cmake_minimum_required(VERSION 3.20)
project(WHT131 VERSION 1.1.3 LANGUAGES C CXX)
if(NOT WIN32)
 message(FATAL_ERROR "Windows is required")
endif()
if(CMAKE_SIZEOF_VOID_P EQUAL 8)
 set(WHT_RUNTIME native/version64.cpp native/exports64.def)
else()
 set(WHT_RUNTIME native/version131.cpp native/exports.def)
endif()
set(CMAKE_CXX_STANDARD 17)
set(CMAKE_CXX_STANDARD_REQUIRED ON)
set(CMAKE_MSVC_RUNTIME_LIBRARY "MultiThreaded$<$<CONFIG:Debug>:Debug>")
add_library(wht_common STATIC native/common.cpp native/license.cpp)
target_include_directories(wht_common PUBLIC native)
target_compile_definitions(wht_common PUBLIC UNICODE _UNICODE WIN32_LEAN_AND_MEAN NOMINMAX)
target_link_libraries(wht_common PUBLIC bcrypt winhttp advapi32)
add_subdirectory(third_party/minhook EXCLUDE_FROM_ALL)
add_library(WHT131 SHARED ${WHT_RUNTIME} native/resources.rc)
target_link_libraries(WHT131 PRIVATE wht_common minhook ws2_32 user32 gdi32 winmm comctl32 shell32)
set_target_properties(WHT131 PROPERTIES OUTPUT_NAME WHT)
add_custom_command(TARGET WHT131 POST_BUILD
 COMMAND powershell -NoProfile -ExecutionPolicy Bypass -File "${CMAKE_CURRENT_SOURCE_DIR}/package-mix.ps1"
 -Dll "$<TARGET_FILE:WHT131>" -Archive "${CMAKE_CURRENT_SOURCE_DIR}/native/assets/tl_overlay.mpq"
 -Output "$<TARGET_FILE_DIR:WHT131>/WHT.mix" VERBATIM)
set_property(TARGET WHT131 APPEND PROPERTY LINK_DEPENDS "${CMAKE_CURRENT_SOURCE_DIR}/native/assets/tl_overlay.mpq" "${CMAKE_CURRENT_SOURCE_DIR}/package-mix.ps1")
add_executable(WHT131Launcher native/launcher131.cpp)
if(CMAKE_SIZEOF_VOID_P EQUAL 4)
add_executable(WHT131ExtractAssets build-tools/extract-assets.cpp)
target_compile_definitions(WHT131ExtractAssets PRIVATE UNICODE _UNICODE WIN32_LEAN_AND_MEAN NOMINMAX)
target_compile_options(WHT131ExtractAssets PRIVATE /W4 /utf-8)
endif()
target_link_libraries(WHT131Launcher PRIVATE wht_common user32 gdi32)
foreach(t wht_common WHT131 WHT131Launcher)
 target_compile_options(${t} PRIVATE /W4 /utf-8 /permissive-)
endforeach()
include(CTest)
if(BUILD_TESTING)
 if(CMAKE_SIZEOF_VOID_P EQUAL 8)
 add_executable(WHT131Tests tests/runtime64-tests.cpp)
 else()
 add_executable(WHT131Tests tests/runtime131-tests.cpp)
 endif()
 target_link_libraries(WHT131Tests PRIVATE wht_common minhook ws2_32 user32 gdi32 comctl32 shell32)
 target_compile_options(WHT131Tests PRIVATE /W4 /utf-8 /permissive-)
 add_test(NAME runtime131-contracts COMMAND WHT131Tests)
 if(CMAKE_SIZEOF_VOID_P EQUAL 4)
 add_executable(WHTMixTests tests/mix-tests.cpp)
 target_compile_options(WHTMixTests PRIVATE /W4 /utf-8)
 target_compile_definitions(WHTMixTests PRIVATE UNICODE _UNICODE WIN32_LEAN_AND_MEAN NOMINMAX)
 add_dependencies(WHTMixTests WHT131)
 add_test(NAME mix131-package-contracts COMMAND WHTMixTests "$<TARGET_FILE_DIR:WHT131>/WHT.mix" "${CMAKE_CURRENT_SOURCE_DIR}/tests/support/Storm.dll" "${CMAKE_CURRENT_SOURCE_DIR}/native/assets/BTNWHT_Rec.blp")
endif()
endif()

add_executable(WHTExternalHud WIN32 native/external_hud_main.cpp)
target_compile_definitions(WHTExternalHud PRIVATE UNICODE _UNICODE WIN32_LEAN_AND_MEAN NOMINMAX)
target_compile_options(WHTExternalHud PRIVATE /W4 /utf-8 /permissive-)
target_link_libraries(WHTExternalHud PRIVATE d3d11 dxgi d2d1 dwrite dcomp user32 ole32 shell32 psapi)
set_target_properties(WHTExternalHud PROPERTIES OUTPUT_NAME "WHT.ExternalHud")
if(BUILD_TESTING)
 add_executable(WHTExternalHudTests tests/external-hud-tests.cpp)
 target_compile_definitions(WHTExternalHudTests PRIVATE UNICODE _UNICODE WIN32_LEAN_AND_MEAN NOMINMAX)
 target_compile_options(WHTExternalHudTests PRIVATE /W4 /utf-8 /permissive-)
 target_link_libraries(WHTExternalHudTests PRIVATE user32)
 add_test(NAME external-hud-contracts COMMAND WHTExternalHudTests)
 add_test(NAME external-hud-dx11 COMMAND WHTExternalHud --self-test)
endif()

if(BUILD_TESTING)
 add_library(WHT131PluginFixture SHARED tests/dll-loader-fixture.cpp)
 target_compile_definitions(WHT131PluginFixture PRIVATE UNICODE _UNICODE WIN32_LEAN_AND_MEAN NOMINMAX)
 add_executable(WHT131DllLoaderTests tests/dll-loader-tests.cpp)
 target_link_libraries(WHT131DllLoaderTests PRIVATE wht_common)
 target_compile_options(WHT131DllLoaderTests PRIVATE /W4 /utf-8 /permissive-)
 add_dependencies(WHT131DllLoaderTests WHT131PluginFixture)
 add_test(NAME dll-loader-contracts COMMAND WHT131DllLoaderTests "$<TARGET_FILE:WHT131PluginFixture>")
endif()

if(BUILD_TESTING)
 add_executable(WHT131MpqLoaderTests tests/mpq-loader-tests.cpp)
 target_link_libraries(WHT131MpqLoaderTests PRIVATE wht_common)
 target_compile_options(WHT131MpqLoaderTests PRIVATE /W4 /utf-8 /permissive-)
 add_test(NAME mpq-loader-contracts COMMAND WHT131MpqLoaderTests "${CMAKE_CURRENT_SOURCE_DIR}/native/assets/tl_overlay.mpq")
