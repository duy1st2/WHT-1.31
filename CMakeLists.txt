cmake_minimum_required(VERSION 3.20)
project(WHT129 VERSION 0.1.0 LANGUAGES C CXX)
option(BUILD_TESTING "Build 1.29 contracts" OFF)
if(NOT WIN32 OR NOT CMAKE_SIZEOF_VOID_P EQUAL 4)
 message(FATAL_ERROR "Use Visual Studio -A Win32; Warcraft 1.29 is x86.")
endif()
set(CMAKE_CXX_STANDARD 17)
set(CMAKE_CXX_STANDARD_REQUIRED ON)
set(CMAKE_MSVC_RUNTIME_LIBRARY "MultiThreaded$<$<CONFIG:Debug>:Debug>")
add_subdirectory(third_party/minhook EXCLUDE_FROM_ALL)
add_library(wht_common STATIC native/common.cpp native/license.cpp)
target_include_directories(wht_common PUBLIC native)
target_compile_definitions(wht_common PUBLIC UNICODE _UNICODE WIN32_LEAN_AND_MEAN NOMINMAX)
target_link_libraries(wht_common PUBLIC bcrypt winhttp advapi32)
add_library(WHT129 SHARED native/version129.cpp native/exports.def native/resources.rc)
target_link_libraries(WHT129 PRIVATE wht_common minhook user32 gdi32 winmm advapi32 ws2_32)
set_target_properties(WHT129 PROPERTIES OUTPUT_NAME WHT)
add_library(WHTAutoload129 SHARED native/autoload129.cpp)
target_compile_definitions(WHTAutoload129 PRIVATE UNICODE _UNICODE WIN32_LEAN_AND_MEAN NOMINMAX)
target_link_options(WHTAutoload129 PRIVATE /NOIMPLIB /NOEXP)
set_target_properties(WHTAutoload129 PROPERTIES OUTPUT_NAME mss32)
find_package(Python3 COMPONENTS Interpreter REQUIRED)
add_custom_command(TARGET WHTAutoload129 POST_BUILD
 COMMAND "${Python3_EXECUTABLE}" "${CMAKE_CURRENT_SOURCE_DIR}/fix-mss32-exports.py" "$<TARGET_FILE:WHTAutoload129>" "${CMAKE_CURRENT_SOURCE_DIR}/native/mss32-129-exports.json" VERBATIM)
set_property(TARGET WHTAutoload129 APPEND PROPERTY LINK_DEPENDS "${CMAKE_CURRENT_SOURCE_DIR}/fix-mss32-exports.py" "${CMAKE_CURRENT_SOURCE_DIR}/native/mss32-129-exports.json")
add_custom_command(TARGET WHT129 POST_BUILD
 COMMAND powershell -NoProfile -ExecutionPolicy Bypass -File "${CMAKE_CURRENT_SOURCE_DIR}/package-mix.ps1"
 -Dll "$<TARGET_FILE:WHT129>" -Archive "${CMAKE_CURRENT_SOURCE_DIR}/native/assets/tl_overlay.mpq"
 -Output "$<TARGET_FILE_DIR:WHT129>/WHT.mix" VERBATIM)
set_property(TARGET WHT129 APPEND PROPERTY LINK_DEPENDS "${CMAKE_CURRENT_SOURCE_DIR}/native/assets/tl_overlay.mpq" "${CMAKE_CURRENT_SOURCE_DIR}/package-mix.ps1")
foreach(t wht_common WHT129 WHTAutoload129)
 target_compile_options(${t} PRIVATE /W4 /utf-8 /permissive-)
endforeach()
include(CTest)
if(BUILD_TESTING)
 add_executable(WHTVersion129Tests tests/version129-tests.cpp)
 target_link_libraries(WHTVersion129Tests PRIVATE wht_common minhook user32 gdi32 winmm advapi32 ws2_32)
 target_compile_options(WHTVersion129Tests PRIVATE /W4 /utf-8)
 add_test(NAME version129-contracts COMMAND WHTVersion129Tests)
 add_executable(WHTMixTests tests/mix-tests.cpp)
 target_compile_options(WHTMixTests PRIVATE /W4 /utf-8)
 target_compile_definitions(WHTMixTests PRIVATE UNICODE _UNICODE WIN32_LEAN_AND_MEAN NOMINMAX)
 add_dependencies(WHTMixTests WHT129)
 add_test(NAME mix129-package-contracts COMMAND WHTMixTests "$<TARGET_FILE_DIR:WHT129>/WHT.mix" "${CMAKE_CURRENT_SOURCE_DIR}/tests/support/Storm.dll" "${CMAKE_CURRENT_SOURCE_DIR}/native/assets/BTNWHT_Rec.blp")
endif()
